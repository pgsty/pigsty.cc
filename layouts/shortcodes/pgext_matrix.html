= "ext-avail-cell--pgdg" }}{{ end -}} {{- if eq $repo "PIGSTY" }}{{ $cls = "ext-avail-cell--pigsty" }}{{ end -}} {{- if eq $repo "CONTRIB" }}{{ $cls = "ext-avail-cell--contrib" }}{{ end -}} {{- else $verDisplay = "MISS" -}} {{- else if eq $state "N/A" -}} {{- $verDisplay = "N/A" -}} {{- else if or (eq $ver "-") (eq $ver "") -}} {{- $verDisplay = "AVAIL" -}} {{- end -}} {{- $label := htmlEscape
